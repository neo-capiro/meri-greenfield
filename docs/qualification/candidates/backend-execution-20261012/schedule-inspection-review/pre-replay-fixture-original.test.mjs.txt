const receipt=(request,expiresAtMs)=>({authorized:true,projectionSha256:request.projectionSha256,callerIdentity:request.callerIdentity,expiresAtMs});
function harness(options={}){
 let time=Date.now();const calls=[];let closes=0;
 const clock=options.clock??(()=>time);
 const desc=options.description??description();
 const connection={withDeadline:options.withDeadline??(async(deadline,fn)=>{calls.push(['deadline',deadline]);return fn()}),withAbortSignal:options.withAbortSignal??(async(signal,fn)=>{calls.push(['abort',signal]);return fn()}),close(){closes++;throw Error('borrowed close')}};
 const scheduleClient={connection,getHandle(id){calls.push(['handle',id]);return{describe:async()=>{calls.push(['describe']);return options.describe?options.describe():desc},delete(){throw Error('mutation')},pause(){throw Error('mutation')},trigger(){throw Error('mutation')}}},list(){throw Error('list')},create(){throw Error('mutation')},close(){closes++;throw Error('borrowed close')}};
 const authorize=options.authorize??(async request=>{calls.push(['authorize',request.phase]);return receipt(request,clock()+40000)});
 const inspector=createTemporalScheduleInspector({scheduleClient,authorize,clock});
 const request={projection:projection(),callerIdentity:caller(),deadlineMs:clock()+50000};
 return{inspector,request,calls,desc,connection,scheduleClient,clock,advance(ms){time+=ms},closes:()=>closes,run(){return inspector.inspect(request)}};
}
const bounded=async promise=>{let timer;try{return await Promise.race([promise,new Promise((_,reject)=>{timer=setTimeout(()=>reject(Error('independent test timeout')),1500)})])}finally{clearTimeout(timer)}};
