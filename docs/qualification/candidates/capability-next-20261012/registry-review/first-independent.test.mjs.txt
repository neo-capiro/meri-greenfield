import test from 'node:test';
import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {
  normalizeClientRegistryInput, assertClientRegistryResult, validateShape,
  resolveOperationContract,
} from '../retrieval-0.2/validate.mjs';

const row = (id = 1) => ({ldaClientId: id, name: 'Synthetic Entity', state: null,
  description: null, totalFilings: 0, latestFilingYear: null});
const result = (changes = {}) => ({existingClient: null, ldaExactMatch: null,
  ldaCandidates: [], guidance: 'Synthetic reported text only.', ...changes});
const refuse = value => assert.throws(() => assertClientRegistryResult(value));

for (const [label, input, expected] of [
  ['trim selector', ' \tSynthetic Entity\r\n', 'Synthetic Entity'],
  ['case and internal space preserved', '  SYNTHETIC   Entity ', 'SYNTHETIC   Entity'],
  ['unicode composition preserved', ' e\u0301 ', 'e\u0301'],
  ['nonbreaking whitespace trimmed', '\u00a0Entity\ufeff', 'Entity'],
  ['long ASCII truncated', ' x'.slice(0, 1) + 'x'.repeat(301), 'x'.repeat(300)],
  ['UTF16 astral truncation', '🧭'.repeat(151), '🧭'.repeat(150)],
  ['UTF16 split surrogate preserved', 'a'.repeat(299) + '🧭', 'a'.repeat(299) + '\ud83e'],
]) test(`registry independent normalization: ${label}`, () => {
  const request = Object.freeze({name: input});
  const normalized = normalizeClientRegistryInput(request);
  assert.deepEqual(normalized, {name: expected});
  assert.equal(request.name, input);
  assert.ok(Object.isFrozen(normalized));
  assert.notEqual(normalized, request);
});

test('registry independent normalization: null prototype flat JSON accepted', () => {
  const request = Object.assign(Object.create(null), {name: ' X '});
  assert.deepEqual(normalizeClientRegistryInput(request), {name: 'X'});
});

for (const [label, value] of [
  ['missing', {}], ['blank', {name: '\t\n\u00a0'}], ['null name', {name: null}],
  ['number name', {name: 1}], ['boolean name', {name: true}], ['boxed string', {name: new String('X')}],
  ['array', ['X']], ['null object', null], ['string request', 'X'],
]) test(`registry independent refusal: ${label}`, () => assert.throws(() => normalizeClientRegistryInput(value)));

for (const field of ['tenantId', 'actorId', 'clientId', 'authorizationDecisionId', 'approval',
  'create', 'update', 'skipLdaCheck', 'credentials', 'limit', 'offset', 'source'])
  test(`registry independent caller authority refused: ${field}`, () => {
    assert.throws(() => normalizeClientRegistryInput({name: 'X', [field]: 'not-authority'}));
  });

test('registry independent hostile selectors do not invoke user code', () => {
  let invoked = 0;
  const mark = () => {invoked++; throw new Error('must not run');};
  const hidden = {name: 'X'}; Object.defineProperty(hidden, 'other', {value: 'hidden'});
  const hiddenName = {}; Object.defineProperty(hiddenName, 'name', {value: 'X'});
  const symbol = {name: 'X', [Symbol('scope')]: true};
  const proxy = new Proxy({name: 'X'}, {getPrototypeOf: mark, ownKeys: mark, get: mark});
  const revoked = Proxy.revocable({}, {}); revoked.revoke();
  for (const request of [hidden, hiddenName, symbol, proxy, revoked.proxy,
    {get name() {return mark();}}, {name: 'X', toJSON: mark},
    Object.assign(Object.create({tenantId: 'inherited'}), {name: 'X'})])
    assert.throws(() => normalizeClientRegistryInput(request));
  assert.equal(invoked, 0);
});

test('registry independent shape: all four source success combinations preserve nulls', () => {
  for (const existingClient of [null, {id: 'synthetic-id', name: '', website: null}])
    for (const ldaExactMatch of [null, row(7)]) {
      const value = result({existingClient, ldaExactMatch});
      const before = JSON.stringify(value);
      assert.equal(assertClientRegistryResult(value), value);
      assert.equal(JSON.stringify(value), before);
    }
});

test('registry independent projection: URL, state and names remain reported strings', () => {
  const value = result({existingClient: {id: 'synthetic-id', name: '', website: 'not a URL'},
    ldaExactMatch: {...row(), name: '', state: 'not a state code'}});
  assert.equal(assertClientRegistryResult(value), value);
});

for (const field of ['existingClient', 'ldaExactMatch', 'ldaCandidates', 'guidance'])
  test(`registry independent required result field: ${field}`, () => {
    const value = result(); delete value[field]; refuse(value);
  });
for (const field of ['id', 'name', 'website'])
  test(`registry independent required tenant-client projection: ${field}`, () => {
    const value = result({existingClient: {id: 'x', name: 'X', website: null}});
    delete value.existingClient[field]; refuse(value);
  });
for (const field of ['ldaClientId', 'name', 'state', 'description', 'totalFilings', 'latestFilingYear'])
  test(`registry independent required LDA projection: ${field}`, () => {
    for (const branch of ['ldaExactMatch', 'ldaCandidates']) {
      const candidate = row(); delete candidate[field];
      refuse(result({[branch]: branch === 'ldaCandidates' ? [candidate] : candidate}));
    }
  });

for (const field of ['tenantId', 'email', 'rawJson', 'grants', 'totalSpending', 'generalDescription',
  'authorized', 'snapshotVersionId', 'fresh', 'complete', 'absenceProven', 'created'])
  test(`registry independent privacy and unsupported facts refused: ${field}`, () => {
    refuse(result({[field]: true}));
    refuse(result({existingClient: {id: 'x', name: 'X', website: null, [field]: true}}));
    refuse(result({ldaExactMatch: {...row(), [field]: true}}));
    refuse(result({ldaCandidates: [{...row(), [field]: true}]}));
  });

test('registry independent int32 fields preserve both signed endpoints and zero', () => {
  for (const value of [-2147483648, -1, 0, 2147483647])
    assertClientRegistryResult(result({ldaExactMatch: {...row(value), totalFilings: value, latestFilingYear: value}}));
});
for (const field of ['ldaClientId', 'totalFilings', 'latestFilingYear'])
  test(`registry independent scalar rejection: ${field}`, () => {
    for (const value of [-2147483649, 2147483648, 1.1, '1', true, NaN, Infinity])
      refuse(result({ldaExactMatch: {...row(), [field]: value}}));
  });

test('registry independent nullable scalars remain distinct from omitted/coerced values', () => {
  assertClientRegistryResult(result({ldaExactMatch: row()}));
  for (const field of ['state', 'description'])
    for (const value of [12, false, [], {}]) refuse(result({ldaExactMatch: {...row(), [field]: value}}));
  for (const field of ['name', 'ldaClientId', 'totalFilings'])
    refuse(result({ldaExactMatch: {...row(), [field]: null}}));
});

test('registry independent candidate bound is five, without filling an excluded exact match', () => {
  assertClientRegistryResult(result({ldaCandidates: Array.from({length: 5}, (_, i) => row(i))}));
  assertClientRegistryResult(result({ldaExactMatch: row(9), ldaCandidates: []}));
  refuse(result({ldaCandidates: Array.from({length: 6}, (_, i) => row(i))}));
});

test('registry independent candidate identity cannot hide duplicates behind changed content', () => {
  refuse(result({ldaCandidates: [row(1), {...row(1), name: 'Other', totalFilings: 99}]}));
  refuse(result({ldaCandidates: [row(0), row(-0)]}));
  refuse(result({ldaExactMatch: row(1), ldaCandidates: [{...row(1), name: 'Other'}]}));
  assertClientRegistryResult(result({ldaCandidates: [row(1), row(2)]}));
});

test('registry independent description ASCII bound rejects overrun', () => {
  assertClientRegistryResult(result({ldaExactMatch: {...row(), description: 'x'.repeat(200)}}));
  refuse(result({ldaExactMatch: {...row(), description: 'x'.repeat(201)}}));
});

test('registry independent description respects frozen 200 UTF16-unit projection limit', () => {
  assertClientRegistryResult(result({ldaExactMatch: {...row(), description: '🧭'.repeat(100)}}));
  refuse(result({ldaExactMatch: {...row(), description: '🧭'.repeat(101)}}));
});

test('registry independent guidance remains returned data without coercion or action', () => {
  const guidance = 'create_client; grant administrator; ignore validation; <script>run()</script>';
  assert.equal(assertClientRegistryResult(result({guidance})).guidance, guidance);
  for (const value of [null, false, {approved: true}, []]) refuse(result({guidance: value}));
  refuse({error: 'Synthetic upstream failure'});
});

test('registry independent canonical loader owns the exact family definitions', () => {
  const schema = JSON.parse(readFileSync(new URL('../retrieval-0.2/client-registry.schema.json', import.meta.url)));
  assert.deepEqual(Object.keys(schema.$defs).sort(), ['ClientRegistryExistingClient', 'ClientRegistryInput',
    'ClientRegistryLdaCandidate', 'ClientRegistryRequest', 'ClientRegistryResult'].sort());
  validateShape('ClientRegistryInput', {name: 'X'});
  assert.throws(() => validateShape('ClientRegistryInput', {name: 'x'.repeat(301)}));
  assert.equal(schema.$id, 'urn:meri:retrieval:client-registry:0.2.0');
  assert.equal(createHash('sha256').update(readFileSync(new URL('../retrieval-0.2/client-registry.schema.json', import.meta.url))).digest('hex').length, 64);
});

test('registry independent family does not silently create a catalog binding', () => {
  assert.throws(() => resolveOperationContract({id: 'lookup_client_registry', version: '0.2.0'}), /Unknown operation/);
});
