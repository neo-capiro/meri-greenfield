import test from 'node:test';import assert from 'node:assert/strict';import {readFile} from 'node:fs/promises';import vm from 'node:vm';
import {fixture,envelope,NOW,id} from '../fixtures.mjs';import {prepare,event,canonicalSha256} from '../binding.mjs';
// Isolate only the existing pure assertion, without invoking the real runner or any SQL.
// No implementation copy or SQL simulator; exact source is retained in Git/receipts.
const source=await readFile(new URL('./run-persistence.mjs',import.meta.url),'utf8');
const text=source.slice(source.indexOf('function assertResultRows('),source.indexOf('export async function runLocalActionPersistence'));
assert.ok(text.startsWith('function assertResultRows('));
const assertRows=vm.runInNewContext(`(${text})`,{assert,prepare,canonicalSha256});
function values(){const f=fixture(),c={lookup:{request:f.request,context:f.context,credential:'synthetic'},expectedChargeMicroUsd:10},b=prepare(f.request,f.context,NOW),observation=envelope(b),resultSha256=canonicalSha256(observation),events=[event(b,'tool.started',NOW),event(b,'tool.completed',NOW,f.context.resultReference)],read={classification:'recommendations',acceptedTasks:false,observation,resultSha256,events},invocationId='owned_'+id(700),reservationId=id(701);const rows={lookups:[{state:'RESULT',payload:b,record:{binding:b,observation,resultSha256,events}}],events:events.map((e,i)=>({kind:i?'terminal':'started',sequence:String(e.sequence),event_id:e.eventId.toLowerCase(),event:e})),invocations:[{invocation_id:invocationId,reservation_id:reservationId,arguments_sha256:b.argumentsSha256,completion:{invocationId,reservationId,argumentsSha256:b.argumentsSha256,resultSha256,outputBytes:4242}}],reservations:[{invocation_id:invocationId,reservation_id:reservationId,state:'SETTLED',hold_micro_usd:'10',charged_micro_usd:'10',output_bytes:'4242'}]};return {c,rows,read};}
test('independent exact correlated local result rows remain admissible as source observations',()=>{const {rows,c,read}=values();assert.doesNotThrow(()=>assertRows(rows,c,read));});
for(const kind of ['output bytes','completion invocation','completion reservation'])test(`independent contradictory ${kind} cannot pass persistence evidence`,()=>{const {rows,c,read}=values();if(kind==='output bytes')rows.reservations[0].output_bytes='1';else if(kind==='completion invocation')rows.invocations[0].completion.invocationId='owned_'+id(799);else rows.invocations[0].completion.reservationId=id(799);assert.throws(()=>assertRows(rows,c,read));});
